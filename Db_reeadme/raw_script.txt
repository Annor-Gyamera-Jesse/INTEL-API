SELECT 
    s.StudentID,
    s.StudentFirstName + ' ' + s.StudentLastName AS StudentName,
    ft.FeeTypeName,
    ft.Amount                  AS FeeExpected,
    sf.AmountPaid              AS WhatWasPaid,
    sf.AmountLeft              AS RemainingBalance,     -- ← this is probably 19.40 or NULL
    CASE 
        WHEN sf.StudentID IS NULL THEN ft.Amount
        WHEN sf.AmountLeft IS NULL THEN 0
        ELSE sf.AmountLeft
    END AS CalculatedOwing     -- ← this matches exactly what your dashboard uses
FROM SchoolManagement.Students s
INNER JOIN SchoolManagement.FeeTypes ft 
    ON s.ClassID = ft.ClassID
LEFT JOIN SchoolManagement.StudentFees sf 
    ON sf.StudentID = s.StudentID 
   AND sf.FeeTypeID = ft.FeeTypeID;

   -- Total expected across ALL FeeTypes for this student
SELECT 
    s.StudentID,

FROM SchoolManagement.Students s
INNER JOIN SchoolManagement.FeeTypes ft ON s.ClassID = ft.ClassID
WHERE s.StudentID = 10029
GROUP BY s.StudentID, s.StudentFirstName, s.StudentLastName;

SELECT 
    ft.FeeTypeName,
    ft.Amount AS Expected,
    COALESCE(sf.AmountPaid, 0) AS Paid,

-- Stored Procedure to Generate Token
DROP PROCEDURE SchoolManagement.GenerateToken
    @UserName VARCHAR(50),
    @Password VARCHAR(100)
AS
BEGIN
    DECLARE @UserID INT;
    DECLARE @Token VARCHAR(255);
    DECLARE @ExpiryDateTime DATETIME;

    -- Check if the provided credentials are valid
    SELECT @UserID = UserID
    FROM SchoolManagement.Users
    WHERE UserName = @UserName AND Password = @Password;

    -- If valid, generate a token
    IF @UserID IS NOT NULL
    BEGIN
        SET @Token = CONVERT(VARCHAR(255), NEWID());
        SET @ExpiryDateTime = DATEADD(HOUR, 1, GETDATE()); -- Token expires in 1 hour

        -- Store the token in the Tokens table
        INSERT INTO SchoolManagement.Tokens (UserID, Token, ExpiryDateTime)
        VALUES (@UserID, @Token, @ExpiryDateTime);

        -- Return the generated token
        SELECT @Token AS Token;
    END
    ELSE
    BEGIN
        -- Return an error or handle invalid credentials
        THROW 50000, 'Invalid credentials.', 1;
    END
END;

    f.name AS ForeignKey,
    OBJECT_NAME(f.parent_object_id) AS ReferencingTable,
    COL_NAME(fc.parent_object_id, fc.parent_column_id) AS ReferencingColumn
FROM 
    sys.foreign_keys AS f
INNER JOIN 
    sys.foreign_key_columns AS fc 
    ON f.OBJECT_ID = fc.constraint_object_id
WHERE 
    OBJECT_NAME(f.referenced_object_id) = 'Students';

    @ClassID VARCHAR(10),
    @CourseName VARCHAR(50)
AS
BEGIN
    SELECT

        e.Class,
        e.SchoolCourse,

        e.ExamsSubDescription,

    WHERE e.Class = @ClassID AND e.SchoolCourse = @CourseName
END

CREATE PROCEDURE GetTeachersAttendanceByFullName
    @TeacherFullName VARCHAR(100) -- This is a single parameter for the full name
AS
BEGIN
    SELECT 
        TA.TeacherFirstName,
        TA.TeacherLastName,
        CASE 
            WHEN TA.EnableSwitch = 1 THEN 'Present'
            ELSE 'Absent'
        END AS AttendanceStatus,
        TA.ClockIN AS ClockInTime,
        TAO.ClockOUT AS ClockOutTime,
        TA.RecDateCreated AS RecordCreatedDate
    FROM 
        SchoolManagement.TeachersAttendance TA
    LEFT JOIN 
        SchoolManagement.TeachersAttendanceOut TAO
    ON 
        TA.TeacherFirstName = TAO.TeacherFirstName
        AND TA.TeacherLastName = TAO.TeacherLastName
    WHERE 
        (@TeacherFullName IS NULL OR (TA.TeacherFirstName + ' ' + TA.TeacherLastName) = @TeacherFullName);
END

CREATE PROCEDURE SchoolManagement.sp_AddStudentFeePayment
    @StudentID INT,
    @FeeTypeID INT,
    @AmountPaid DECIMAL(10,2),
    @ClassID VARCHAR(10),
    @TermID INT,
    @UserID INT,
    @PaymentMethod VARCHAR(MAX),
    @Note VARCHAR(MAX) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @FeeTypeAmount DECIMAL(10,2);
    SELECT @FeeTypeAmount = Amount 
    FROM SchoolManagement.FeeTypes
    WHERE FeeTypeID = @FeeTypeID AND ClassID = @ClassID;

    IF @FeeTypeAmount IS NULL
    BEGIN
        RAISERROR('Invalid FeeTypeID or ClassID.', 16, 1);
        RETURN;
    END

    -- Step 1: Sum of previous payments for this student + fee + term
    DECLARE @TotalPaidBefore DECIMAL(10,2) = 0;
    SELECT @TotalPaidBefore = SUM(ISNULL(AmountPaid, 0))
    FROM SchoolManagement.StudentFees
    WHERE StudentID = @StudentID AND FeeTypeID = @FeeTypeID AND TermID = @TermID;

    DECLARE @TotalExpectedPayment DECIMAL(10,2) = @TotalPaidBefore + @AmountPaid;

    IF @TotalExpectedPayment > @FeeTypeAmount
    BEGIN
        DECLARE @RemainingBalance DECIMAL(10,2) = @FeeTypeAmount - @TotalPaidBefore;
        DECLARE @RemainingBalanceStr VARCHAR(50) = CAST(@RemainingBalance AS VARCHAR(50));
        RAISERROR('Payment exceeds the required fee. Remaining balance is ₵%s', 16, 1, @RemainingBalanceStr);
        RETURN;
    END

    -- Step 2: Calculate AmountLeft for the new row
    DECLARE @NewAmountLeft DECIMAL(10,2) = @FeeTypeAmount - @TotalExpectedPayment;

    DECLARE @PaymentStatus VARCHAR(20) = 
        CASE WHEN @NewAmountLeft = 0 THEN 'Fully Paid' ELSE 'Not Fully Paid' END;

    -- Step 3: Insert new payment record
    INSERT INTO SchoolManagement.StudentFees (
        StudentID, FeeTypeID, StudentName, FeeTypeName, ClassID,
        AmountPaid, AmountLeft, PaymentDate, DueDate, Note,
        UserID, TermID, PaymentMethod, PaymentStatus
    )
    SELECT 
        s.StudentID,
        @FeeTypeID,
        CONCAT(s.StudentFirstName, ' ', s.StudentLastName),
        ft.FeeTypeName,
        @ClassID,
        @AmountPaid,
        @NewAmountLeft,
        GETDATE(),
        NULL,
        @Note,
        @UserID,
        @TermID,
        @PaymentMethod,
        @PaymentStatus
    FROM SchoolManagement.Students s
    JOIN SchoolManagement.FeeTypes ft ON ft.FeeTypeID = @FeeTypeID
    WHERE s.StudentID = @StudentID;

    -- Step 4: Update all rows that have no balance left to "Fully Paid"
    IF @NewAmountLeft = 0
    BEGIN
        UPDATE SchoolManagement.StudentFees
        SET PaymentStatus = 'Fully Paid'
        WHERE StudentID = @StudentID
          AND FeeTypeID = @FeeTypeID
          AND TermID = @TermID
          AND PaymentStatus = 'Not Fully Paid'
          AND AmountLeft = 0;
    END
END;

-- =============================================
-- 1. Assign Fees to All Students For New Term
-- =============================================
CREATE OR ALTER PROCEDURE SchoolManagement.sp_AssignFeesForNewTerm
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @CurrentTermID INT, @PreviousTermID INT;

    -- Get current and previous term
    SELECT @CurrentTermID = TermID FROM SchoolManagement.SchoolTerm WHERE IsCurrentTerm = 1;
    
    SELECT TOP 1 @PreviousTermID = TermID
    FROM SchoolManagement.SchoolTerm
    WHERE TermID < @CurrentTermID
    ORDER BY TermID DESC;

    IF @PreviousTermID IS NULL RETURN;

    -- Create temp table for assignments
    IF OBJECT_ID('tempdb..#TempAssignments') IS NOT NULL DROP TABLE #TempAssignments;

    CREATE TABLE #TempAssignments (
        StudentID INT,
        FeeTypeID INT,
        StudentName VARCHAR(MAX),
        FeeTypeName VARCHAR(255),
        ClassID VARCHAR(MAX),
        CarriedAmount DECIMAL(10,2),
        NewFeeAmount DECIMAL(10,2),
        TotalAmountLeft DECIMAL(10,2),
        Note VARCHAR(MAX),
        UserID INT
    );

    -- Prepare data with carry-over + current term fees
    INSERT INTO #TempAssignments (
        StudentID, FeeTypeID, StudentName, FeeTypeName, ClassID,
        CarriedAmount, NewFeeAmount, TotalAmountLeft, Note, UserID
    )
    SELECT 
        s.StudentID,
        ft.FeeTypeID,
        CONCAT(s.StudentFirstName, ' ', s.StudentLastName),
        ft.FeeTypeName,
        s.ClassID,
        ISNULL(prev.AmountLeft, 0) AS Carried,
        ISNULL(ft.Amount, 0) AS NewFee,
        ISNULL(prev.AmountLeft, 0) + ISNULL(ft.Amount, 0) AS TotalDue,
        CASE 
            WHEN ISNULL(prev.AmountLeft, 0) > 0 AND ISNULL(ft.Amount, 0) > 0 THEN
                CONCAT('₵', CAST(prev.AmountLeft AS VARCHAR), ' carried over from Term ', @PreviousTermID, 
                       '. New term fee ₵', CAST(ft.Amount AS VARCHAR), ' added.')
            WHEN ISNULL(prev.AmountLeft, 0) > 0 AND ISNULL(ft.Amount, 0) = 0 THEN
                CONCAT('₵', CAST(prev.AmountLeft AS VARCHAR), ' carried over from Term ', @PreviousTermID,
                       '. ⚠️ New term fee missing or 0.')
            WHEN ISNULL(prev.AmountLeft, 0) = 0 AND ISNULL(ft.Amount, 0) > 0 THEN
                CONCAT('New term fee ₵', CAST(ft.Amount AS VARCHAR), ' assigned.')
            ELSE
                '⚠️ No fee or carry-over found for this student.'
        END,
        s.UserID
    FROM SchoolManagement.Students s
    JOIN SchoolManagement.FeeTypes ft 
        ON LTRIM(RTRIM(ft.ClassID)) = LTRIM(RTRIM(s.ClassID))
    LEFT JOIN (
        SELECT StudentID, FeeTypeID, SUM(AmountLeft) AS AmountLeft
        FROM SchoolManagement.StudentFees
        WHERE TermID = @PreviousTermID AND AmountLeft > 0
        GROUP BY StudentID, FeeTypeID
    ) prev ON prev.StudentID = s.StudentID AND prev.FeeTypeID = ft.FeeTypeID;

    -- Insert new fee records for students who don’t have them yet
    INSERT INTO SchoolManagement.StudentFees (
        StudentID, FeeTypeID, StudentName, FeeTypeName, ClassID,
        AmountPaid, AmountLeft, PaymentDate, DueDate, Note,
        UserID, TermID, PaymentMethod, PaymentStatus
    )
    SELECT 
        t.StudentID,
        t.FeeTypeID,
        t.StudentName,
        t.FeeTypeName,
        t.ClassID,
        0.00,
        t.TotalAmountLeft,
        GETDATE(),
        NULL,
        t.Note,
        t.UserID,
        @CurrentTermID,
        'AutoAssign',
        CASE WHEN t.TotalAmountLeft = 0 THEN 'Fully Paid' ELSE 'Not Fully Paid' END
    FROM #TempAssignments t
    WHERE NOT EXISTS (
        SELECT 1 
        FROM SchoolManagement.StudentFees f
        WHERE f.StudentID = t.StudentID 
          AND f.FeeTypeID = t.FeeTypeID 
          AND f.TermID = @CurrentTermID
    );

    -- Log inserted records
    INSERT INTO SchoolManagement.FeeAssignmentLog (
        StudentID, FeeTypeID, TermID, ActionType, PreviousAmountLeft, NewAmountLeft, Note
    )
    SELECT 
        t.StudentID,
        t.FeeTypeID,
        @CurrentTermID,
        'Inserted',
        t.CarriedAmount,
        t.TotalAmountLeft,
        t.Note
    FROM #TempAssignments t
    WHERE NOT EXISTS (
        SELECT 1 
        FROM SchoolManagement.StudentFees f
        WHERE f.StudentID = t.StudentID 
          AND f.FeeTypeID = t.FeeTypeID 
          AND f.TermID = @CurrentTermID
    );

    -- Update any wrong fee record (auto-heal)
    UPDATE f
    SET 
        f.AmountLeft = t.TotalAmountLeft,
        f.AmountPaid = 0.00,
        f.Note = t.Note,
        f.PaymentMethod = 'AutoAssign',
        f.PaymentDate = GETDATE(),
        f.PaymentStatus = CASE WHEN t.TotalAmountLeft = 0 THEN 'Fully Paid' ELSE 'Not Fully Paid' END
    FROM SchoolManagement.StudentFees f
    JOIN #TempAssignments t 
        ON f.StudentID = t.StudentID AND f.FeeTypeID = t.FeeTypeID AND f.TermID = @CurrentTermID
    WHERE f.AmountLeft <> t.TotalAmountLeft;

    -- Log updated records
    INSERT INTO SchoolManagement.FeeAssignmentLog (
        StudentID, FeeTypeID, TermID, ActionType, PreviousAmountLeft, NewAmountLeft, Note
    )
    SELECT 
        f.StudentID,
        f.FeeTypeID,
        f.TermID,
        'Updated',
        f.AmountLeft,
        t.TotalAmountLeft,
        t.Note
    FROM SchoolManagement.StudentFees f
    JOIN #TempAssignments t 
        ON f.StudentID = t.StudentID AND f.FeeTypeID = t.FeeTypeID AND f.TermID = @CurrentTermID
    WHERE f.AmountLeft <> t.TotalAmountLeft;

    -- Cleanup
    DROP TABLE #TempAssignments;
END;

-- 2. Carry Over Unpaid Balances to New Term
-- =============================================
DROP PROCEDURE IF EXISTS SchoolManagement.sp_CarryOverUnpaidFees;

CREATE OR ALTER PROCEDURE SchoolManagement.sp_CarryOverUnpaidFees
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @CurrentTermID INT, @PreviousTermID INT;

    -- Get the current term
    SELECT @CurrentTermID = TermID 
    FROM SchoolManagement.SchoolTerm 
    WHERE IsCurrentTerm = 1;

    -- Get the previous term
    SELECT TOP 1 @PreviousTermID = TermID
    FROM SchoolManagement.SchoolTerm
    WHERE TermID < @CurrentTermID
    ORDER BY TermID DESC;

    -- Exit if no previous term found
    IF @PreviousTermID IS NULL RETURN;

    --------------------------------------------------------------------------------
    -- Step 1: Carry forward unpaid balances to the new term
    --------------------------------------------------------------------------------
    INSERT INTO SchoolManagement.StudentFees (
        StudentID, FeeTypeID, StudentName, FeeTypeName, ClassID,
        AmountPaid, AmountLeft, PaymentDate, DueDate, Note,
        UserID, TermID, PaymentMethod, PaymentStatus
    )
    SELECT 
        sf.StudentID,
        sf.FeeTypeID,
        sf.StudentName,
        sf.FeeTypeName,
        s.ClassID,
        0.00,
        sf.AmountLeft,
        GETDATE(),
        NULL,
        'Auto carried forward from previous term',
        sf.UserID,
        @CurrentTermID,
        'AutoCarry',
        'Not Fully Paid'
    FROM SchoolManagement.StudentFees sf
    JOIN SchoolManagement.Students s ON s.StudentID = sf.StudentID
    WHERE sf.AmountLeft > 0 
      AND sf.TermID = @PreviousTermID
      -- Prevent inserting duplicates of same student + fee + term
      AND NOT EXISTS (
            SELECT 1 FROM SchoolManagement.StudentFees sf2
            WHERE sf2.StudentID = sf.StudentID 
              AND sf2.FeeTypeID = sf.FeeTypeID 
              AND sf2.TermID = @CurrentTermID
      );

    --------------------------------------------------------------------------------
    -- Step 2: Record rollover in StudentFeeAccount table
    --------------------------------------------------------------------------------
    INSERT INTO SchoolManagement.StudentFeeAccount (
        StudentID, StudentName, TermID,
        CurrentClassID, PreviousClassID,
        CurrentFeeTypeID, CurrentFeeTypeName, CurrentFeeAmount,
        CurrentAmountPaid, CurrentAmountLeft, CurrentPaymentStatus,
        PreviousFeeTypeID, PreviousFeeTypeName, PreviousFeeAmount,
        PreviousAmountPaid, PreviousAmountLeft, PreviousPaymentStatus,
        RollOverOutstandingBalance, Note, UserID
    )
    SELECT 
        sf.StudentID,
        sf.StudentName,
        @CurrentTermID,
        s.ClassID,
        (
            SELECT TOP 1 PreviousClassID 
            FROM SchoolManagement.StudentClassHistory h 
            WHERE h.StudentID = sf.StudentID 
            ORDER BY h.UpdateDate DESC
        ),
        sf.FeeTypeID,
        sf.FeeTypeName,
        ft.Amount,
        0.00,
        sf.AmountLeft,
        'Not Fully Paid',
        sf.FeeTypeID,
        sf.FeeTypeName,
        ft.Amount,
        sf.AmountPaid,
        sf.AmountLeft,
        sf.PaymentStatus,
        sf.AmountLeft,
        'Auto rolled from previous term',
        sf.UserID
    FROM SchoolManagement.StudentFees sf
    JOIN SchoolManagement.Students s ON s.StudentID = sf.StudentID
    JOIN SchoolManagement.FeeTypes ft ON ft.FeeTypeID = sf.FeeTypeID
    WHERE sf.TermID = @PreviousTermID
      AND sf.AmountLeft > 0
      -- Avoid duplicate account rollovers
      AND NOT EXISTS (
          SELECT 1 
          FROM SchoolManagement.StudentFeeAccount a
          WHERE a.StudentID = sf.StudentID 
            AND a.TermID = @CurrentTermID
      );
END;

-- =============================================
CREATE OR ALTER TRIGGER trg_OnTermChange_CarryOver
ON SchoolManagement.SchoolTerm
AFTER UPDATE
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (
        SELECT 1 
        FROM inserted i JOIN deleted d ON i.TermID = d.TermID
        WHERE i.IsCurrentTerm = 1 AND d.IsCurrentTerm = 0
    )
    BEGIN

CREATE OR ALTER TRIGGER trg_AfterStudentPromotion
ON SchoolManagement.StudentClassHistory
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @CurrentTermID INT;

    BEGIN TRY
        SELECT @CurrentTermID = TermID FROM SchoolManagement.SchoolTerm WHERE IsCurrentTerm = 1;

        -- ✅ Log: Trigger fired
        INSERT INTO SchoolManagement.TriggerLogs
        (TriggerName, TableAffected, ActionType, Message, CreatedAt)
        VALUES
        ('trg_AfterStudentPromotion', 'StudentClassHistory', 'AFTER INSERT', 
         'Trigger fired successfully. TermID = ' + CAST(@CurrentTermID AS VARCHAR), GETDATE());

        -- ✅ Log: Inserting fees
        INSERT INTO SchoolManagement.TriggerLogs
        (TriggerName, TableAffected, ActionType, Message, StudentID, TermID)
        SELECT 
            'trg_AfterStudentPromotion', 'StudentFees', 'INSERT',
            'Preparing to insert auto-generated fees after promotion.',
            i.StudentID, @CurrentTermID
        FROM inserted i;

        -- ✅ Insert new student fees
        INSERT INTO SchoolManagement.StudentFees (
            StudentID, FeeTypeID, StudentName, FeeTypeName, ClassID,
            AmountPaid, AmountLeft, PaymentDate, DueDate, Note,
            UserID, TermID, PaymentMethod, PaymentStatus
        )
        SELECT 
            s.StudentID,
            f.FeeTypeID,
            CONCAT(s.StudentFirstName, ' ', s.StudentLastName),
            f.FeeTypeName,
            s.ClassID,
            0.00,
            f.Amount,
            GETDATE(),
            NULL,
            'Auto-assigned after promotion',
            s.UserID,
            @CurrentTermID,
            'AutoAssign',
            'Not Fully Paid'
        FROM inserted i
        JOIN SchoolManagement.Students s ON s.StudentID = i.StudentID
        JOIN SchoolManagement.FeeTypes f ON f.ClassID = s.ClassID
        WHERE NOT EXISTS (
            SELECT 1 FROM SchoolManagement.StudentFees sf
            WHERE sf.StudentID = s.StudentID 
              AND sf.FeeTypeID = f.FeeTypeID 
              AND sf.TermID = @CurrentTermID
              AND sf.ClassID = s.ClassID  -- ✅ NEW: prevents stale class blocking
        );

        -- ✅ Insert into StudentFeeAccount
        INSERT INTO SchoolManagement.StudentFeeAccount (
            StudentID, StudentName, TermID,
            CurrentClassID, PreviousClassID,
            CurrentFeeTypeID, CurrentFeeTypeName, CurrentFeeAmount,
            CurrentAmountPaid, CurrentAmountLeft, CurrentPaymentStatus,
            PreviousFeeTypeID, PreviousFeeTypeName, PreviousFeeAmount,
            PreviousAmountPaid, PreviousAmountLeft, PreviousPaymentStatus,
            RollOverOutstandingBalance, Note, UserID
        )
        SELECT
            s.StudentID,
            CONCAT(s.StudentFirstName, ' ', s.StudentLastName),
            @CurrentTermID,
            s.ClassID,
            i.PreviousClassID,
            f.FeeTypeID,
            f.FeeTypeName,
            f.Amount,
            0.00,
            f.Amount,
            'Not Fully Paid',
            pf.FeeTypeID,
            pf.FeeTypeName,
            pf.Amount,
            ISNULL(pf.AmountPaid, 0),
            ISNULL(pf.AmountLeft, pf.Amount),
            ISNULL(pf.PaymentStatus, 'Not Fully Paid'),
            ISNULL(pf.AmountLeft, pf.Amount),
            'Auto-generated on promotion',
            s.UserID
        FROM inserted i
        JOIN SchoolManagement.Students s ON s.StudentID = i.StudentID
        JOIN SchoolManagement.FeeTypes f ON f.ClassID = s.ClassID
        LEFT JOIN (
            SELECT *
            FROM (
                SELECT sf.StudentID, sf.FeeTypeID, sf.FeeTypeName, f.Amount,
                       sf.AmountPaid, sf.AmountLeft, sf.PaymentStatus,
                       ROW_NUMBER() OVER (PARTITION BY sf.StudentID ORDER BY sf.TermID DESC) AS rn
                FROM SchoolManagement.StudentFees sf
                JOIN SchoolManagement.FeeTypes f ON f.FeeTypeID = sf.FeeTypeID
                WHERE sf.TermID < @CurrentTermID
            ) latest
            WHERE latest.rn = 1
        ) pf ON pf.StudentID = s.StudentID
        WHERE NOT EXISTS (
            SELECT 1 FROM SchoolManagement.StudentFeeAccount a
            WHERE a.StudentID = s.StudentID 
              AND a.TermID = @CurrentTermID
              AND a.CurrentClassID = s.ClassID  -- ✅ NEW: prevents stale class blocking
        );

        -- ✅ Log: Fee summary insert success
        INSERT INTO SchoolManagement.TriggerLogs

            'Fee summary inserted successfully for promoted student.',
            i.StudentID, @CurrentTermID
        FROM inserted i;

        PRINT 'Trigger completed successfully. TermID = ' + CAST(@CurrentTermID AS VARCHAR);

    END TRY

        DECLARE @AnyStudentID INT = (SELECT TOP 1 StudentID FROM inserted);
        DECLARE @EmailSent BIT = 0;

            EXEC SchoolManagement.SendTriggerErrorEmail

        VALUES
        ('trg_AfterStudentPromotion', 'StudentClassHistory', 'ERROR',
         'Trigger failed: ' + @ErrorMsg, @AnyStudentID, @CurrentTermID, GETDATE(), @EmailSent);

CREATE OR ALTER PROCEDURE SchoolManagement.GetStudentFeeSummaryReport
    @AllTerms BIT = 0,
    @TermID INT = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT 
        s.StudentFirstName + ' ' + s.StudentLastName AS StudentName,
        s.ClassID AS CurrentClass,
        ISNULL(sf.TermID, ISNULL(fa.TermID, @TermID)) AS CurrentTerm,

        -- Fee Type
        ft.FeeTypeName AS CurrentFeeType,
        ft.Amount AS CurrentAmount,

        -- Payments
        ISNULL(sf.AmountPaid, 0) AS CurrentPaid,
        ISNULL(ft.Amount - ISNULL(sf.AmountPaid, 0), ft.Amount) AS CurrentLeft,

        -- Previous Class
        sch.PreviousClassID AS PreviousClass,

        -- Previous Fee Info (optional)
        fa.PreviousFeeTypeName AS PreviousFeeType,
        fa.PreviousFeeAmount AS PreviousAmount,
        fa.PreviousAmountPaid AS PreviousPaid,
        fa.PreviousAmountLeft AS PreviousLeft,

        -- Reason
        CASE 
            WHEN sf.AmountPaid IS NULL THEN 
                CONCAT('₵', FORMAT(ft.Amount, 'N2'), ' assigned from FeeTypes table (no payment made yet)')
            WHEN fa.RollOverOutstandingBalance IS NOT NULL THEN 
                CONCAT(
                    '₵', FORMAT(fa.RollOverOutstandingBalance, 'N2'), ' carried over from Term ', ISNULL((fa.TermID - 1), 0), 
                    '. New term fee ₵', FORMAT(ft.Amount, 'N2'), ' added.'
                )
            ELSE 
                CONCAT('New term fee ₵', FORMAT(ft.Amount, 'N2'), ' assigned.')
        END AS Reason

    FROM SchoolManagement.Students s

    -- Join FeeTypes by class
    INNER JOIN SchoolManagement.FeeTypes ft ON ft.ClassID = s.ClassID

    -- Get payment summary from StudentFees (if any)
    LEFT JOIN (
        SELECT StudentID, FeeTypeID, TermID,
               SUM(AmountPaid) AS AmountPaid
        FROM SchoolManagement.StudentFees
        WHERE (@AllTerms = 1 OR TermID = @TermID)
        GROUP BY StudentID, FeeTypeID, TermID
    ) sf ON sf.StudentID = s.StudentID AND sf.FeeTypeID = ft.FeeTypeID

    -- Get previous class
    LEFT JOIN (
        SELECT StudentID, MAX(PreviousClassID) AS PreviousClassID
        FROM SchoolManagement.StudentClassHistory
        GROUP BY StudentID
    ) sch ON sch.StudentID = s.StudentID

    -- Get fee account for previous info
    LEFT JOIN SchoolManagement.StudentFeeAccount fa 
        ON fa.StudentID = s.StudentID 
        AND fa.CurrentFeeTypeID = ft.FeeTypeID 
        AND (@AllTerms = 1 OR fa.TermID = @TermID)

    -- Optional filter if needed
    WHERE s.EnableSwitch = 1

    ORDER BY s.ClassID, StudentName, CurrentFeeType;
END

CREATE PROCEDURE SchoolManagement.FixStudentFeeAmounts
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @StudentID INT, @FeeTypeID INT, @TermID INT;
    DECLARE @TotalOwed DECIMAL(10,2);
    DECLARE @TotalPaidSoFar DECIMAL(10,2);
    DECLARE @FeeID INT, @AmountPaid DECIMAL(10,2);

    DECLARE student_cursor CURSOR FOR
    SELECT DISTINCT StudentID, FeeTypeID, TermID
    FROM SchoolManagement.StudentFees;

    OPEN student_cursor;
    FETCH NEXT FROM student_cursor INTO @StudentID, @FeeTypeID, @TermID;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        -- Step 1: Calculate TOTAL OWED for this student+fee+term (from assignment rows)
        SELECT @TotalOwed = SUM(ISNULL(AmountLeft, 0))
        FROM SchoolManagement.StudentFees
        WHERE StudentID = @StudentID AND FeeTypeID = @FeeTypeID AND TermID = @TermID
              AND AmountPaid = 0;

        -- Reset paid tracker
        SET @TotalPaidSoFar = 0;

        -- Step 2: Go row-by-row, in chronological order
        DECLARE payment_cursor CURSOR FOR
        SELECT FeeID, AmountPaid
        FROM SchoolManagement.StudentFees
        WHERE StudentID = @StudentID AND FeeTypeID = @FeeTypeID AND TermID = @TermID
        ORDER BY RecDateCreated, FeeID;

        OPEN payment_cursor;
        FETCH NEXT FROM payment_cursor INTO @FeeID, @AmountPaid;

        WHILE @@FETCH_STATUS = 0
        BEGIN
            -- Step 3: If assignment row (AmountPaid = 0), update it with full total owed
            IF @AmountPaid = 0
            BEGIN
                UPDATE SchoolManagement.StudentFees
                SET AmountLeft = @TotalOwed
                WHERE FeeID = @FeeID;
            END
            ELSE
            BEGIN
                SET @TotalPaidSoFar += @AmountPaid;
                DECLARE @NewLeft DECIMAL(10,2) = @TotalOwed - @TotalPaidSoFar;
                IF @NewLeft < 0 SET @NewLeft = 0;

                UPDATE SchoolManagement.StudentFees
                SET AmountLeft = @NewLeft
                WHERE FeeID = @FeeID;
            END

            FETCH NEXT FROM payment_cursor INTO @FeeID, @AmountPaid;
        END

        CLOSE payment_cursor;
        DEALLOCATE payment_cursor;

        FETCH NEXT FROM student_cursor INTO @StudentID, @FeeTypeID, @TermID;
    END

    CLOSE student_cursor;
    DEALLOCATE student_cursor;
END

);

/*online*/
CREATE TRIGGER trg_AfterInsert_PaymentsOtherFees
ON SchoolManagement.PaymentsOtherFees
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRY
        DECLARE @FeeID INT,
                @AmountPaid DECIMAL(10,2),
                @PaymentMethodName VARCHAR(255),
                @PaymentMethodID INT,
                @BankNumber VARCHAR(255),
                @SystemTransferID INT,
                @UserID INT;

        -- Get inserted values (assuming single row insert)
        SELECT 
            @FeeID = FeeID,
            @AmountPaid = AmountPaid,
            @PaymentMethodName = PaymentMethod,
            @UserID = UserID
        FROM inserted;

        -- Get PaymentMethod details
        SELECT 
            @PaymentMethodID = PaymentMethodID,
            @BankNumber = BankNumber
        FROM SchoolManagement.PaymentMethods
        WHERE MethodName = @PaymentMethodName;

        -- Generate a unique SystemTransferID (can also be NEWID if GUID is preferred)
        SET @SystemTransferID = ABS(CHECKSUM(NEWID()));

        -- Insert into Bank table
        INSERT INTO SchoolManagement.Bank
        (
            PaymentMethodID,
            BankNumber,
            MethodName,
            AmountTransferred,
            SystemTransferID,
            UserID,
            Remarks,
            AmountInHand
        )
        VALUES
        (
            @PaymentMethodID,
            @BankNumber,
            @PaymentMethodName,
            @AmountPaid,
            @SystemTransferID,
            @UserID,
            'Auto-transfer from PaymentsOtherFees',
            @AmountPaid
        );

        -- Update the PaymentsOtherFees row to reflect transfer
        UPDATE SchoolManagement.PaymentsOtherFees
        SET 
            SystemTransferStatus = 'Success',
            SystemTransferID = @SystemTransferID
        WHERE FeeID = @FeeID;

        -- Log success
        INSERT INTO SchoolManagement.PaymentsOtherFeesTriggerLogs
        (TriggerName, ActionTaken, AmountProcessed, Status, CreatedAt)
        VALUES
        ('trg_AfterInsert_PaymentsOtherFees',
         CONCAT('Transferred ', @AmountPaid, ' via ', @PaymentMethodName),
         @AmountPaid,
         'Success',
         GETDATE());
    END TRY
    BEGIN CATCH
        -- Log failure
        INSERT INTO SchoolManagement.PaymentsOtherFeesTriggerLogs
        (TriggerName, ActionTaken, AmountProcessed, Status, ErrorMessage, CreatedAt)
        VALUES
        ('trg_AfterInsert_PaymentsOtherFees',
         'Insert into Bank Failed',
         @AmountPaid,
         'Failed',
         ERROR_MESSAGE(),
         GETDATE());

        -- Update PaymentsOtherFees with error
        UPDATE SchoolManagement.PaymentsOtherFees
        SET 
            SystemTransferStatus = 'Failed',
            TransferErrorMessage = ERROR_MESSAGE()
        WHERE FeeID = @FeeID;
    END CATCH
END;

CREATE OR ALTER TRIGGER trg_AfterInsert_Bank
ON SchoolManagement.Bank
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;

    -- Calculate the total AmountTransferred per MethodName including newly inserted rows
    ;WITH TotalTransfers AS
    (
        SELECT
            b.MethodName,
            SUM(b.AmountTransferred) AS TotalAmount
        FROM SchoolManagement.Bank b
        INNER JOIN (SELECT DISTINCT MethodName FROM inserted) i
            ON b.MethodName = i.MethodName
        GROUP BY b.MethodName
    )
    MERGE SchoolManagement.BankBalances AS target
    USING TotalTransfers AS source
    ON target.MethodName = source.MethodName
    WHEN MATCHED THEN
        UPDATE SET TotalAmount = source.TotalAmount
    WHEN NOT MATCHED THEN
        INSERT (MethodName, TotalAmount) VALUES (source.MethodName, source.TotalAmount);

    -- Log the insert action
    INSERT INTO SchoolManagement.BankLog (
        BankID,
        MethodName,
        AmountTransferred,
        AmountInHand,
        TransferDate,
        ActionType,
        UserID
    )
    SELECT
        BankID,
        MethodName,
        AmountTransferred,
        AmountInHand,
        TransferDate,
        'INSERT',
        UserID
    FROM inserted;
END;

CREATE OR ALTER PROCEDURE SchoolManagement.sp_GetDirectorBankStatement
AS
BEGIN
    SET NOCOUNT ON;

    -- ── 1. Current balance per payment method ──────────────────────────────
    SELECT
        MethodName,
        SUM(AmountTransferred)  AS CurrentBalance,
        COUNT(*)                AS TotalTransactions,
        MAX(TransferDate)       AS LastActivityDate
    FROM SchoolManagement.Bank
    GROUP BY MethodName;

    -- ── 2. Full money trail — every GHS in and out ─────────────────────────
    -- This tells the director: where did money come from, where did it go
    SELECT
        b.BankID,
        b.TransferDate                                              AS Date,
        b.MethodName                                               AS PaymentMethod,
        b.AmountTransferred                                        AS Amount,
        b.AmountInHand                                             AS BalanceAfter,
        b.Remarks                                                  AS Description,
        CASE
            WHEN b.AmountTransferred > 0 THEN 'Money In'
            WHEN b.AmountTransferred < 0 THEN 'Money Out'
            ELSE 'No Change'
        END                                                        AS Direction,
        CASE
            WHEN b.Remarks LIKE '%Auto-transfer from StudentFees%'
                THEN 'Student Fee Payment'
            WHEN b.Remarks LIKE '%Auto-transfer%'
                THEN 'Automatic Transfer'
            WHEN b.Remarks LIKE '%carried%'
                THEN 'Fee Carry-Over'
            WHEN b.Remarks LIKE '%salary%' OR b.Remarks LIKE '%Salary%'
                THEN 'Salary Payment'
            WHEN b.Remarks LIKE '%expense%' OR b.Remarks LIKE '%Expense%'
                THEN 'School Expense'
            WHEN b.Remarks LIKE '%Manual Deposit%'
                THEN 'Manual Deposit'
            WHEN b.Remarks LIKE '%transfer%' OR b.Remarks LIKE '%Transfer%'
                THEN 'Fund Transfer'
            WHEN b.Remarks LIKE '%withdrawal%' OR b.Remarks LIKE '%Withdrawal%'
                THEN 'Withdrawal'
            WHEN b.Remarks LIKE '%Initial%'
                THEN 'Opening Balance'
            ELSE 'Other Transaction'
        END                                                        AS Category,
        -- Plain English explanation for the director
        CASE
            WHEN b.AmountTransferred > 0 AND b.Remarks LIKE '%StudentFees%'
                THEN 'A student fee payment of GH₵' + CAST(ABS(b.AmountTransferred) AS VARCHAR) +
                     ' was received and deposited into ' + b.MethodName + '.'
            WHEN b.AmountTransferred > 0 AND b.Remarks LIKE '%Manual Deposit%'
                THEN 'GH₵' + CAST(ABS(b.AmountTransferred) AS VARCHAR) +
                     ' was manually deposited into ' + b.MethodName + '.'
            WHEN b.AmountTransferred < 0 AND b.Remarks LIKE '%salary%'
                THEN 'GH₵' + CAST(ABS(b.AmountTransferred) AS VARCHAR) +
                     ' was paid out as salary from ' + b.MethodName + '.'
            WHEN b.AmountTransferred < 0
                THEN 'GH₵' + CAST(ABS(b.AmountTransferred) AS VARCHAR) +
                     ' was deducted from ' + b.MethodName + '. ' +
                     ISNULL('Reason: ' + b.Remarks, 'No reason recorded.')
            WHEN b.AmountTransferred > 0
                THEN 'GH₵' + CAST(ABS(b.AmountTransferred) AS VARCHAR) +
                     ' was added to ' + b.MethodName + '. ' +
                     ISNULL('Note: ' + b.Remarks, '')
            ELSE 'Transaction recorded with no change in balance.'
        END                                                        AS PlainEnglishExplanation,
        u.FullName                                                 AS RecordedBy
    FROM SchoolManagement.Bank b
    LEFT JOIN SchoolManagement.Users u ON b.UserID = u.UserID
    ORDER BY b.TransferDate DESC;

    -- ── 3. Grand summary — one number the director trusts ──────────────────
    SELECT
        ISNULL(SUM(CASE WHEN AmountTransferred > 0
                        THEN AmountTransferred ELSE 0 END), 0)    AS TotalMoneyIn,
        ISNULL(SUM(CASE WHEN AmountTransferred < 0
                        THEN ABS(AmountTransferred) ELSE 0 END), 0) AS TotalMoneyOut,
        ISNULL(SUM(AmountTransferred), 0)                         AS NetBalance,
        COUNT(CASE WHEN AmountTransferred > 0 THEN 1 END)         AS TotalDeposits,
        COUNT(CASE WHEN AmountTransferred < 0 THEN 1 END)         AS TotalWithdrawals,
        MIN(TransferDate)                                          AS FirstTransaction,
        MAX(TransferDate)                                          AS LastTransaction
    FROM SchoolManagement.Bank;

    -- ── 4. Expense breakdown — what money left the school and why ──────────
    SELECT
        ec.CategoryName                                            AS ExpenseCategory,
        COUNT(e.ExpenseID)                                         AS NumberOfExpenses,
        SUM(e.Amount)                                              AS TotalSpent,
        MAX(e.ExpenseDate)                                         AS LastExpenseDate,
        -- Tell the director in plain English
        'The school spent GH₵' + CAST(ISNULL(SUM(e.Amount),0) AS VARCHAR) +
        ' on ' + ec.CategoryName + ' across ' +
        CAST(COUNT(e.ExpenseID) AS VARCHAR) + ' transaction(s).'    AS Summary
    FROM SchoolManagement.Expenses e
    JOIN SchoolManagement.ExpenseCategories ec
        ON e.CategoryID = ec.CategoryID
    GROUP BY ec.CategoryName
    ORDER BY TotalSpent DESC;

END;
GO

CREATE PROCEDURE SchoolManagement.TransferStudentFeesToBank
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @TransferID INT;
    
    -- Start Transaction
    BEGIN TRANSACTION;

    BEGIN TRY
        -- Generate a new Transfer ID
        SELECT @TransferID = ISNULL(MAX(SystemTransferID), 0) + 1 FROM SchoolManagement.StudentFees;
        
        -- Insert into Bank table
        INSERT INTO SchoolManagement.Bank (PaymentMethodID, BankNumber, MethodName, AmountTransferred, SystemTransferID)
        SELECT 
            pm.PaymentMethodID,
            pm.BankNumber,
            pm.MethodName,
            sf.AmountPaid,
            @TransferID
        FROM SchoolManagement.StudentFees sf
        JOIN SchoolManagement.PaymentMethods pm ON sf.PaymentMethod = pm.MethodName
        WHERE sf.SystemTransferStatus = 'New';

        -- Update StudentFees to mark as completed
        UPDATE SchoolManagement.StudentFees
        SET 
            SystemTransferStatus = 'Completed', 
            SystemTransferID = @TransferID
        WHERE SystemTransferStatus = 'New';

        -- Commit transaction
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        -- Capture Error Message
        DECLARE @ErrorMessage NVARCHAR(MAX) = ERROR_MESSAGE();
        
        -- Rollback transaction
        ROLLBACK TRANSACTION;

        -- Store Error Message in StudentFees Table
        UPDATE SchoolManagement.StudentFees
        SET TransferErrorMessage = @ErrorMessage
        WHERE SystemTransferStatus = 'New';

        -- Re-throw error
        THROW;
    END CATCH;
END;

ALTER PROCEDURE SchoolManagement.TransferStudentFeesToBank
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRANSACTION;

    BEGIN TRY
        -- Merge new amounts with existing records in the Bank table
        MERGE INTO SchoolManagement.Bank AS Target
        USING (
            SELECT 
                sf.PaymentMethod AS PaymentMethodID,
                pm.BankNumber,
                pm.MethodName,
                SUM(sf.AmountPaid) AS TotalAmountTransferred
            FROM SchoolManagement.StudentFees sf
            JOIN SchoolManagement.PaymentMethods pm ON sf.PaymentMethod = pm.MethodName
            WHERE sf.SystemTransferStatus = 'New'
            GROUP BY sf.PaymentMethod, pm.BankNumber, pm.MethodName
        ) AS Source
        ON Target.PaymentMethodID = Source.PaymentMethodID AND Target.BankNumber = Source.BankNumber
        WHEN MATCHED THEN
            -- Update existing records by adding the new amount
            UPDATE SET Target.AmountTransferred = Target.AmountTransferred + Source.TotalAmountTransferred
        WHEN NOT MATCHED THEN
            -- Insert new records if they don't exist
            INSERT (PaymentMethodID, BankNumber, MethodName, AmountTransferred, TransferDate, SystemTransferID)
            VALUES (Source.PaymentMethodID, Source.BankNumber, Source.MethodName, Source.TotalAmountTransferred, GETDATE(), 
                    (SELECT ISNULL(MAX(SystemTransferID), 0) + 1 FROM SchoolManagement.Bank));

        -- Update StudentFees to mark as completed
        UPDATE sf
        SET sf.SystemTransferStatus = 'Completed'
        FROM SchoolManagement.StudentFees sf
        JOIN SchoolManagement.PaymentMethods pm ON sf.PaymentMethod = pm.MethodName
        WHERE sf.SystemTransferStatus = 'New';

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        THROW;
    END CATCH;
END;

CREATE TRIGGER SchoolManagement.trg_ErrorGuard_StudentFees
ON SchoolManagement.StudentFees
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @Snapshot NVARCHAR(MAX);
    DECLARE @badSID   INT;
    DECLARE @badUID   INT;

    BEGIN TRY

        -- Validate: AmountPaid cannot be negative
        IF EXISTS (SELECT 1 FROM inserted WHERE AmountPaid < 0)
        BEGIN
            SELECT TOP 1 @badSID = StudentID, @badUID = UserID
            FROM inserted WHERE AmountPaid < 0;

            SELECT @Snapshot = (
                SELECT TOP 1 FeeID, StudentID, FeeTypeID, AmountPaid, AmountLeft, TermID, UserID
                FROM inserted WHERE AmountPaid < 0
                FOR JSON PATH, WITHOUT_ARRAY_WRAPPER
            );

            EXEC SchoolManagement.sp_LogError
                @TriggerName       = 'trg_ErrorGuard_StudentFees',
                @TableAffected     = 'StudentFees',
                @OperationType     = 'INSERT/UPDATE',
                @ErrorNumber       = 50001,
                @ErrorMessage      = 'Negative AmountPaid detected. Operation blocked.',
                @AffectedStudentID = @badSID,
                @AffectedUserID    = @badUID,
                @SnapshotData      = @Snapshot;

            ROLLBACK TRANSACTION;
            RETURN;
        END

        -- Validate: AmountLeft cannot be negative
        IF EXISTS (SELECT 1 FROM inserted WHERE AmountLeft < 0)
        BEGIN
            SELECT TOP 1 @badSID = StudentID, @badUID = UserID
            FROM inserted WHERE AmountLeft < 0;

            SELECT @Snapshot = (
                SELECT TOP 1 FeeID, StudentID, FeeTypeID, AmountPaid, AmountLeft, TermID, UserID
                FROM inserted WHERE AmountLeft < 0
                FOR JSON PATH, WITHOUT_ARRAY_WRAPPER
            );

            EXEC SchoolManagement.sp_LogError
                @TriggerName       = 'trg_ErrorGuard_StudentFees',
                @TableAffected     = 'StudentFees',
                @OperationType     = 'INSERT/UPDATE',
                @ErrorNumber       = 50002,
                @ErrorMessage      = 'Negative AmountLeft detected. Operation blocked.',
                @AffectedStudentID = @badSID,
                @AffectedUserID    = @badUID,
                @SnapshotData      = @Snapshot;

            ROLLBACK TRANSACTION;
            RETURN;
        END

        -- Validate: StudentID must exist in Students
        IF EXISTS (
            SELECT 1 FROM inserted i
            WHERE NOT EXISTS (
                SELECT 1 FROM SchoolManagement.Students s WHERE s.StudentID = i.StudentID
            )
        )
        BEGIN
            SELECT TOP 1 @badSID = StudentID FROM inserted;

            SELECT @Snapshot = (
                SELECT TOP 1 FeeID, StudentID, FeeTypeID, AmountPaid, AmountLeft, TermID, UserID
                FROM inserted
                FOR JSON PATH, WITHOUT_ARRAY_WRAPPER
            );

            EXEC SchoolManagement.sp_LogError
                @TriggerName       = 'trg_ErrorGuard_StudentFees',
                @TableAffected     = 'StudentFees',
                @OperationType     = 'INSERT',
                @ErrorNumber       = 50003,
                @ErrorMessage      = 'StudentID does not exist in Students table. Operation blocked.',
                @AffectedStudentID = @badSID,
                @SnapshotData      = @Snapshot;

            ROLLBACK TRANSACTION;
            RETURN;
        END

    END TRY
    BEGIN CATCH
        DECLARE @ErrMsg1 NVARCHAR(2000) = ERROR_MESSAGE();
        DECLARE @ErrNum1 INT = ERROR_NUMBER();
        DECLARE @ErrSev1 INT = ERROR_SEVERITY();
        DECLARE @ErrSta1 INT = ERROR_STATE();
        EXEC SchoolManagement.sp_LogError @TriggerName = 'trg_ErrorGuard_StudentFees', @TableAffected = 'StudentFees', @OperationType = 'INSERT/UPDATE/DELETE', @ErrorNumber = @ErrNum1, @ErrorMessage = @ErrMsg1, @ErrorSeverity = @ErrSev1, @ErrorState = @ErrSta1;
    END CATCH
END;
GO

                @ErrorMessage  = 'Negative AmountPaid on PaymentsOtherFees. Operation blocked.',
                @SnapshotData  = @Snapshot;

            ROLLBACK TRANSACTION;
            RETURN;
        END

        -- Validate: StudentID must exist
        IF EXISTS (
            SELECT 1 FROM inserted i
            WHERE NOT EXISTS (
                SELECT 1 FROM SchoolManagement.Students s WHERE s.StudentID = i.StudentID
            )
        )
        BEGIN
            SELECT TOP 1 @orphanSID = StudentID FROM inserted;

            SELECT @Snapshot = (
                SELECT TOP 1 StudentID, AmountPaid, PaymentMethod
                FROM inserted
                FOR JSON PATH, WITHOUT_ARRAY_WRAPPER
            );

            EXEC SchoolManagement.sp_LogError
                @TriggerName       = 'trg_ErrorGuard_PaymentsOtherFees',
                @TableAffected     = 'PaymentsOtherFees',
                @OperationType     = 'INSERT',
                @ErrorNumber       = 50021,
                @ErrorMessage      = 'StudentID does not exist in Students. Operation blocked.',
                @AffectedStudentID = @orphanSID,
                @SnapshotData      = @Snapshot;

            ROLLBACK TRANSACTION;
            RETURN;
        END

    END TRY
    BEGIN CATCH
        DECLARE @ErrMsg3 NVARCHAR(2000) = ERROR_MESSAGE();
        DECLARE @ErrNum3 INT = ERROR_NUMBER();
        DECLARE @ErrSev3 INT = ERROR_SEVERITY();
        DECLARE @ErrSta3 INT = ERROR_STATE();
        EXEC SchoolManagement.sp_LogError @TriggerName = 'trg_ErrorGuard_PaymentsOtherFees', @TableAffected = 'PaymentsOtherFees', @OperationType = 'INSERT/UPDATE/DELETE', @ErrorNumber = @ErrNum3, @ErrorMessage = @ErrMsg3, @ErrorSeverity = @ErrSev3, @ErrorState = @ErrSta3;
    END CATCH
END;
GO

-- ============================================================
-- STEP 9: QUERY PROCEDURE — sp_GetSystemSafetyLog
-- ============================================================

IF OBJECT_ID('SchoolManagement.sp_GetSystemSafetyLog', 'P') IS NOT NULL
    DROP PROCEDURE SchoolManagement.sp_GetSystemSafetyLog;
GO

CREATE PROCEDURE SchoolManagement.sp_GetSystemSafetyLog
    @TableName  VARCHAR(100) = NULL,
    @FromDate   DATE         = NULL,
    @ToDate     DATE         = NULL,
    @StudentID  INT          = NULL,
    @TopN       INT          = 100
AS
BEGIN
    SET NOCOUNT ON;
    SELECT TOP (@TopN)
        ErrorID,
        TriggerName,
        TableAffected,
        OperationType,
        ErrorNumber,
        ErrorMessage,
        AffectedStudentID,
        AffectedUserID,
        SnapshotData,
        LoggedAt
    FROM SchoolManagement.SystemSafetyLog
    WHERE
        (@TableName IS NULL OR TableAffected     = @TableName)
        AND (@StudentID IS NULL OR AffectedStudentID = @StudentID)
        AND (@FromDate  IS NULL OR CAST(LoggedAt AS DATE) >= @FromDate)
        AND (@ToDate    IS NULL OR CAST(LoggedAt AS DATE) <= ISNULL(@ToDate, CAST(GETDATE() AS DATE)))
    ORDER BY LoggedAt DESC;
END;
GO
